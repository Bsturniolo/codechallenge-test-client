12class SnakeBot: 

13 """Resuelve cada ``your_turn`` y manda la dirección elegida."""

15 def __init__(self, logger): 

16 """Recibe el ``GameLogger`` donde se anotan las acciones.""" 

17 self.logger = logger 

18 self.shooter = Shooter()

20 @staticmethod 

21 def multipliers(data): 

22 """Multiplicadores ``(propio, rival)`` según nuestro lado (v4).""" 

23 mine, theirs = 'multiplier_1', 'multiplier_2' 

24 if data['side'] != 'A': 

25 mine, theirs = theirs, mine 

26 return data.get(mine), data.get(theirs)

28 @staticmethod 

29 def random_direction(): 

30 """Dirección al azar, solo si no hay ninguna decisión posible.""" 

31 return choice(list(DIRS))

33 def build_planner(self, board, head, data): 

34 """Arma el ``MovePlanner`` del turno a partir de los datos. 

35 

36 El dígito objetivo se lee del tablero original; después se marca 

37 la zona de explosión de las bombas (v8) para que el planificador 

38 las evite. 

39 """ 

40 own = Side.from_letter(data['side']) 

41 opp = Side.from_letter(opposite(data['side'])) 

42 mine, theirs = self.multipliers(data) 

43 if mine is not None: 

44 print('multiplier: us={} rival={}'.format(mine, theirs)) 

45 target = board.target_digit() or STARTING_DIGIT 

46 keep = {own.head, own.body, opp.head, opp.body, WALL_CHAR} 

47 return MovePlanner(board.mark_danger(keep), head, own, opp, target, 

48 mine or 1, data.get('remaining_moves')) 

49 

50 def _plan(self, data): 

51 """Dirección elegida por el planner para los datos de un turno.""" 

52 print(data['board']) 

53 board = Board.from_string(data['board']) 

54 head = board.find(data['side']) 

55 if head is None: 

56 return self.random_direction() 

57 planner = self.build_planner(board, head, data) 

58 return planner.choose()[0] or self.random_direction() 

59 

60 def decide(self, data): 

61 """Elige la dirección; ante cualquier error juega al azar. 

62 

63 Un fallo inesperado (por ejemplo un carácter nuevo en el tablero 

64 tras una regla nueva) no debe hacer perder el turno ni cortar la 

65 partida. 

66 """ 

67 try: 

68 return self._plan(data) 

69 except Exception as error: 

70 print('error al decidir, se juega al azar: {!r}'.format(error)) 

71 return self.random_direction()

78 def shot_state(self, data): 

79 """Datos del turno que usa el ``Shooter``, según nuestro lado.""" 

80 mine, theirs = 'score_1', 'score_2' 

81 if data['side'] != 'A': 

82 mine, theirs = theirs, mine 

83 return { 

84 'game_id': data['game_id'], 

85 'score': self._number(data, mine), 

86 'rival_score': self._number(data, theirs), 

87 'own_moves_left': self._number(data, 'remaining_moves') // 2, 

88 'multiplier': self.multipliers(data)[0] or 1, 

89 }

91 def shot_direction(self, data): 

92 """Dirección de disparo (v9) o ``None``; nunca lanza excepciones.""" 

93 try: 

94 board = Board.from_string(data['board']) 

95 head = board.find(data['side']) 

96 if head is None: 

97 return None 

98 opp = Side.from_letter(opposite(data['side'])) 

99 return self.shooter.decide( 

100 board, head, opp, self.shot_state(data)) 

101 except Exception as error: 

102 print('error al evaluar el disparo: {!r}'.format(error)) 

103 return None

116 async def handle_turn(self, websocket, request_data): 

117 """Dispara si conviene; si no, decide y envía el movimiento."""
